a=int(input("Enter 1st number:"))
b=int(input("Enter 2nd number:"))
c=input("Enter a operator:")
if c=="+":
    print("Addition =", a + b)
elif c=="-":
     print("Subtraction =", a - b)
elif c=="*":
    print("Multiplication =", a * b)
elif c=="/":
    print("Division =", a / b)
elif c=="//":
    print("floor Division =", a // b)
elif c=="%":
    print("Remainder =", a % b)
else:
    print("please input a valid operator.")
